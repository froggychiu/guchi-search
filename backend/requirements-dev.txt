#
# Deliberately NOT `-r requirements.txt`. Production builds on python:3.11
# (see Dockerfile), where the pinned sqlalchemy==2.0.36 is fine, but that
# version fails to import on Python 3.13+ — a dev machine on a newer Python
# cannot run the tests with the production pin. Everything here is floored
# rather than pinned so the suite runs on whatever interpreter is available;
# production pinning stays in requirements.txt.
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements-dev.txt
#   .venv/bin/python tests/test_search.py
#   .venv/bin/python tests/test_highlight.py

# from it — ingest still indexes into Meilisearch.
